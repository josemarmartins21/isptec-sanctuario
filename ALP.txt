programa {

    leia(tx)

  }

programa {
  funcao inicio() {
    real si, sf, ti, tf, dt, ds, vm

    escreva("Digite o ponto de partida: ")
    leia(si)
    escreva("Digite o ponto de chegada: ")
    leia(sf)

    ds = sf - si

    escreva("Digite o tempo de partida: ")
    leia(ti)
    escreva("Digite o tempo de chegada: ")
    leia(tf)

    dt = tf - ti
    vm = ds / dt

    escreva("A velocidade média é: ", vm)

  }
}
